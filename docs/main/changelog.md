**Автоматический контроль уникальности:**
```sql
-- Функция триггера для проверки уникальности таблица-проблема
CREATE OR REPLACE FUNCTION mcl.check_table_problem_uniqueness()
RETURNS TRIGGER AS $$
DECLARE
    v_problem_id INTEGER;
BEGIN
    -- Получаем problem_id из solution_id
    SELECT problem_id INTO v_problem_id 
    FROM table_problem_solutions 
    WHERE id = NEW.solution_id;
    
    -- Проверяем уникальность таблица-проблема
    IF EXISTS (
        SELECT 1 FROM mssql_tables_problems_mapping m
        JOIN table_problem_solutions s ON m.solution_id = s.id
        WHERE m.table_id = NEW.table_id 
        AND s.problem_id = v_problem_id
        AND m.id != COALESCE(NEW.id, 0)
    ) THEN
        RAISE EXCEPTION 'Таблица % уже имеет проблему %. Уникальность нарушена.', 
            NEW.table_id, v_problem_id;
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;
```
